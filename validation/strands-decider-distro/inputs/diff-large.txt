+@dataclass(frozen=True)
+class ChangeSignal:
+    path: str
+    touches_runtime: bool
+    touches_tests: bool
+    risk: str
+
+def classify_path(path: str) -> ChangeSignal:
+    runtime = path.startswith("src/") or path.startswith("app/")
+    tests = "test" in path
+    risk = "high" if runtime and not tests else "medium" if runtime else "low"
+    return ChangeSignal(path, runtime, tests, risk)
+
+def select_suites(signals: Iterable[ChangeSignal]) -> list[str]:
+    suites = []
+    if any(s.touches_runtime for s in signals):
+        suites.append("unit")
+    if any(s.risk == "high" for s in signals):
+        suites.append("integration")
+    if any(s.path.endswith(".yml") for s in signals):
+        suites.append("config")
+    return suites or ["smoke"]
